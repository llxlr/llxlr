            q1 := 0: 𝔹;

            // 叠加态
            q0 := H(q0);
            q1 := H(q1);
            q2 := H(q2);

            // Oracle: 标记 |110⟩（q0=1, q1=1, q2=0）
            // 由于没有量子 if/索引，Oracle 需要手动用门电路实现

            // CPHASE 或其他多比特相位门也不支持...

            return (measure(q0), measure(q1), measure(q2));
